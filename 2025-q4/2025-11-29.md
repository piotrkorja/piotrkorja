#!/bin/bash

  

# Create media directory

mkdir -p ~/media

  

# Counter for processed files

count=0

  

# Find all media files and process them

sudo find /Users/Shared/SharedGoogleDrive -type f \( \

    -iname "*.jpg" -o -iname "*.jpeg" -o -iname "*.png" -o -iname "*.gif" \

    -o -iname "*.heic" -o -iname "*.heif" -o -iname "*.webp" -o -iname "*.bmp" \

    -o -iname "*.tiff" -o -iname "*.tif" -o -iname "*.mp4" -o -iname "*.mov" \

    -o -iname "*.avi" -o -iname "*.mkv" -o -iname "*.m4v" -o -iname "*.mpg" \

    -o -iname "*.mpeg" -o -iname "*.wmv" -o -iname "*.flv" -o -iname "*.webm" \

    -o -iname "*.3gp" \) 2>/dev/null | while read -r file; do

  

    # Get creation date (birth time on macOS)

    birth_date=$(sudo stat -f "%SB" -t "%Y-%m" "$file" 2>/dev/null)

  

    if [ -z "$birth_date" ]; then

        # Fallback to modification date if birth date unavailable

        birth_date=$(sudo stat -f "%Sm" -t "%Y-%m" "$file" 2>/dev/null)

    fi

  

    # Extract year and month

    year=$(echo "$birth_date" | cut -d'-' -f1)

    month=$(echo "$birth_date" | cut -d'-' -f2)

  

    # Determine quarter

    if [ "$month" -ge 1 ] && [ "$month" -le 3 ]; then

        quarter="q1"

    elif [ "$month" -ge 4 ] && [ "$month" -le 6 ]; then

        quarter="q2"

    elif [ "$month" -ge 7 ] && [ "$month" -le 9 ]; then

        quarter="q3"

    else

        quarter="q4"

    fi

  

    # Create quarter directory

    target_dir="$HOME/media/${year}-${quarter}"

    mkdir -p "$target_dir"

  

    # Get just the filename

    filename=$(basename "$file")

  

    # Handle duplicate filenames by adding a number

    target_file="$target_dir/$filename"

    if [ -f "$target_file" ]; then

        base="${filename%.*}"

        ext="${filename##*.}"

        counter=1

        while [ -f "$target_dir/${base}_${counter}.${ext}" ]; do

            ((counter++))

        done

        target_file="$target_dir/${base}_${counter}.${ext}"

    fi

  

    # Move the file

    mv "$file" "$target_file"

    ((count++))

    echo "Moved: $filename -> ${year}-${quarter}/"

done

  

echo ""

echo "Total files moved: $count"

**➜**  **media**

sudo find /Users/Shared/SharedGoogleDrive -type f \
    -regex '.*(\.zip|\.rar|\.tar|\.tar\.gz|\.tgz|\.7z)$' -print0 | \
    while IFS= read -r -d $'\0' archive_file; do

    echo "Processing: $archive_file"
    DIR=$(dirname "$archive_file")

    # Use sudo for extraction commands since the find command runs as root and may have created files accessible only to root.
    case "$archive_file" in
        *.zip) sudo unzip -o "$archive_file" -d "$DIR" ;;
        *.rar) sudo unrar x -o+ "$archive_file" "$DIR" ;;
        *.tar|*.tar.gz|*.tgz) sudo tar -xf "$archive_file" -C "$DIR" ;;
        *.7z) sudo 7z x "$archive_file" -o"$DIR" -y ;;
        *) echo "Skipping unknown archive type: $archive_file" ;;
    esac

done
